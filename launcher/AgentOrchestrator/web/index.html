I added chart.js and wired it into index.html:

export function drawChart(svg, days) {
  const max = Math.max(...days.map(d => d.high));
  const points = days.map((d, i) => `${i * 50},${100 - (d.high / max) * 90}`);
  svg.innerHTML = `<polyline points="${points.join(' ')}" fill="none" />`;
}
